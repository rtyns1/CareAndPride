- These are plain C# classes that represent real world things in your school.
- They have private setters-- so that no accidental changes of things like student ID in the UI, and only methods inside this class can change the state.
- This is a form of encapsulation.
- Domain entites use GUIDS for good reasno-- they are globally unique, not like integer IDS. 
- Guids are harder to guess than incremental integers.
- They have private constructurs::-- because EF core needs paramaterless constructor to hydrate objects from the database.
- By not making it private we tell the outside world  "You cannot create a User using new User(). You MUST use the User.Create() factory method."
- This enforces our business rules (e.g., requiring an email with '@') at the point of creation.

- This si an example ::

``` csharp

{
    public Guid Id { get; private set; }
    public string Username { get; private set; }
    public string Email { get; private set; }
    public string PasswordHash { get; private set; }
    public string FullName { get; private set; }
    public UserRole Role { get; private set; }
    public bool IsApproved { get; private set; }
    public bool IsActive { get; private set; }
    public DateTime CreatedAtUtc { get; private set; }
    public string? TwoFactorSecret { get; private set; }
    public DateTime? LastLoginAtUtc { get; private set; }

    // Navigation properties (for EF Core relationships)
    public virtual ICollection<TeacherSubject> TeacherSubjects { get; private set; }
    public virtual ICollection<ClassTeacher> ClassTeachers { get; private set; }

    // Private constructor for EF Core
    private User() { }

    // Factory method to create a new User
    public static User Create(string username, string email, string fullName, UserRole role)
    {
        // Business rules enforced here
        if (string.IsNullOrWhiteSpace(username))
            throw new ArgumentException("Username is required.");
        if (!email.Contains('@'))
            throw new ArgumentException("Invalid email format.");

        return new User
        {
            Id = Guid.NewGuid(),
            Username = username,
            Email = email,
            FullName = fullName,
            Role = role,
            IsApproved = false,  // Default: must be approved by Admin
            IsActive = true,
            CreatedAtUtc = DateTime.UtcNow,
            PasswordHash = string.Empty
        };
    }

    // Methods that change state (encapsulation)
    public void SetPasswordHash(string passwordHash) { ... }
    public void Approve() { ... }
    public void Reject() { ... }
    public void RecordLogin() { ... }
}

public class CreateStudentCommandHandler : IRequestHandler<CreateStudentCommand, Guid>
{
    private readonly IStudentRepository _studentRepository;
    private readonly IClassRepository _classRepository;

    public CreateStudentCommandHandler(IStudentRepository studentRepository, IClassRepository classRepository)
    {
        _studentRepository = studentRepository;
        _classRepository = classRepository;
    }

    public async Task<Guid> Handle(CreateStudentCommand request, CancellationToken cancellationToken)
    {
        // 1. Validate that the class exists
        var classExists = await _classRepository.ExistsAsync(request.ClassId);
        if (!classExists)
            throw new ClassNotFoundException(request.ClassId);

        // 2. Create the Domain entity
        var student = Student.Create(
            request.FirstName,
            request.LastName,
            request.DateOfBirth,
            request.AdmissionNumber,
            request.ClassId,
            request.MedicalConditions
        );

        // 3. Save to the database
        await _studentRepository.AddAsync(student);

        // 4. Return the ID
        return student.Id;
    }
}
```

        RuleFor(x => x.LastName)

            .MaximumLength(50).WithMessage("Last name must not exceed 50 characters.");

        RuleFor(x => x.DateOfBirth)

            .GreaterThan(DateTime.UtcNow.AddYears(-120)).WithMessage("Date of birth is too old.");

            .NotEmpty().WithMessage("Class ID is required.");
    }
}

 - These rules can change based on the UI or API versions.
 - Keeping validation in Applicatin keeps domain pure and stable.

public class StudentRepository : IStudentRepository
{
    private readonly AppDbContext _context;

    public StudentRepository(AppDbContext context)
    {
        _context = context;
    }

    public async Task<Student> GetByIdAsync(Guid id)
    {
        return await _context.Students
            .Include(s => s.Class)
            .Include(s => s.Grades)
            .FirstOrDefaultAsync(s => s.Id == id);
    }

    public async Task<IEnumerable<Student>> GetByClassIdAsync(Guid classId)
    {
        return await _context.Students
            .Where(s => s.ClassId == classId && !s.IsArchived)
            .OrderBy(s => s.LastName)
            .ToListAsync();
    }

    public async Task<IEnumerable<Student>> GetByTeacherIdAsync(Guid teacherId)
    {
        // Complex query: Get students in classes where this teacher is a Class Teacher OR Subject Teacher
        var classIds = await _context.ClassTeachers
            .Where(ct => ct.TeacherId == teacherId)
            .Select(ct => ct.ClassId)
            .ToListAsync();

        var subjectClassIds = await _context.TeacherSubjects
            .Where(ts => ts.TeacherId == teacherId)
            .Select(ts => ts.ClassId)
            .ToListAsync();

        var allClassIds = classIds.Union(subjectClassIds).Distinct();

        return await _context.Students
            .Where(s => allClassIds.Contains(s.ClassId) && !s.IsArchived)
            .OrderBy(s => s.LastName)
            .ToListAsync();
    }

    public async Task AddAsync(Student student)
    {
        await _context.Students.AddAsync(student);
        await _context.SaveChangesAsync();
    }

    public async Task UpdateAsync(Student student)
    {
        _context.Students.Update(student);
        await _context.SaveChangesAsync();
    }

    public async Task DeleteAsync(Guid id)
    {
        var student = await GetByIdAsync(id);
        if (student != null)
        {
            // Soft delete: Archive instead of hard delete
            student.Archive();
            _context.Students.Update(student);
            await _context.SaveChangesAsync();
        }
    }

    public async Task<bool> ExistsAsync(Guid id)
    {
        return await _context.Students.AnyAsync(s => s.Id == id);
    }
}
```

- **Why do repositories live in infrastrucutre?**
- The Domain says: " I need a way to find students by class"
- Infrastructure actually goes to the database and fethes them.
- This keeps the Domain clean and testable. You can test the Domain rules without a real Database.

public class FileSystemStorageService : IFileStorageService
{
    private readonly string _rootPath;
    private readonly ILogger<FileSystemStorageService> _logger;

    public FileSystemStorageService(IConfiguration configuration, ILogger<FileSystemStorageService> logger)
    {
        _rootPath = configuration["FileStorage:RootPath"] ?? "C:\\SchoolFiles\\";
        _logger = logger;

        // Ensure the root directory exists
        if (!Directory.Exists(_rootPath))
            Directory.CreateDirectory(_rootPath);
    }

    public async Task<string> SaveFileAsync(Stream fileStream, string fileName, string subFolder)
    {
        try
        {
            // 1. Generate a unique filename (GUID)
            var extension = Path.GetExtension(fileName);
            var uniqueFileName = $"{Guid.NewGuid()}{extension}";

            // 2. Create the subfolder path
            var folderPath = Path.Combine(_rootPath, subFolder);
            if (!Directory.Exists(folderPath))
                Directory.CreateDirectory(folderPath);

            // 3. Build the full path
            var filePath = Path.Combine(folderPath, uniqueFileName);

            // 4. Save the file
            using (var fileStreamOutput = new FileStream(filePath, FileMode.Create))
            {
                await fileStream.CopyToAsync(fileStreamOutput);
            }

            // 5. Return the relative path (to store in the database)
            var relativePath = Path.Combine(subFolder, uniqueFileName);
            _logger.LogInformation("File saved: {FilePath}", relativePath);

            return relativePath;
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Failed to save file: {FileName}", fileName);
            throw new FileStorageException($"Failed to save file: {fileName}", ex);
        }
    }

    public async Task<Stream> GetFileAsync(string filePath)
    {
        try
        {
            var fullPath = Path.Combine(_rootPath, filePath);
            if (!File.Exists(fullPath))
                throw new FileNotFoundException($"File not found: {filePath}");

            var stream = new FileStream(fullPath, FileMode.Open, FileAccess.Read);
            return await Task.FromResult(stream);
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Failed to retrieve file: {FilePath}", filePath);
            throw;
        }
    }

    public async Task DeleteFileAsync(string filePath)
    {
        try
        {
            var fullPath = Path.Combine(_rootPath, filePath);
            if (File.Exists(fullPath))
            {
                File.Delete(fullPath);
                _logger.LogInformation("File deleted: {FilePath}", filePath);
            }
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Failed to delete file: {FilePath}", filePath);
            throw;
        }

        await Task.CompletedTask;
    }

    public bool FileExists(string filePath)
    {
        var fullPath = Path.Combine(_rootPath, filePath);
        return File.Exists(fullPath);
    }
}
```

### **3.3. Global Exception Handler**
-This is the "Safety Net." It catches any unhandled exception, logs it, and returns a friendly error message to the client.
- AN example would be the ExceptionMiddleware.cs::

```
public class ExceptionMiddleware
{
    private readonly RequestDelegate _next;
    private readonly ILogger<ExceptionMiddleware> _logger;

    public ExceptionMiddleware(RequestDelegate next, ILogger<ExceptionMiddleware> logger)
    {
        _next = next;
        _logger = logger;
    }

    public async Task InvokeAsync(HttpContext context)
    {
        try
        {
            await _next(context);
        }
        catch (DomainException ex)
        {
            _logger.LogWarning(ex, "Domain exception occurred");
            context.Response.StatusCode = StatusCodes.Status400BadRequest;
            context.Response.ContentType = "application/json";
            await context.Response.WriteAsync(JsonSerializer.Serialize(new
            {
                error = ex.Message,
                type = ex.GetType().Name
            }));
        }
        catch (UnauthorizedAccessException ex)
        {
            _logger.LogWarning(ex, "Unauthorized access");
            context.Response.StatusCode = StatusCodes.Status403Forbidden;
            context.Response.ContentType = "application/json";
            await context.Response.WriteAsync(JsonSerializer.Serialize(new
            {
                error = "You do not have permission to perform this action."
            }));
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Unhandled exception");
            context.Response.StatusCode = StatusCodes.Status500InternalServerError;
            context.Response.ContentType = "application/json";
            await context.Response.WriteAsync(JsonSerializer.Serialize(new
            {
                error = "An unexpected error occurred. The administrator has been notified."
            }));
        }
    }
}
```

- **Why the Exception Handler Lives in API?**
- The API is the outermost layer. It is the last line of defense.
- The exception handler catches errors from all lower layers (Domain, Application, Infrastructure) and translates them into HTTP responses.
- This ensures the UI never crashes. It always receives a structured error message.
